<configuration>
  <appSettings>
    <add key="Cliente" value="CLIENTE_A"/>
    <add key="NombreServicio"
         value="ServicioPDF_CLIENTE_A"/>
    <add key="DisplayName"
         value="Servicio PDF - Cliente A"/>
    <add key="RutaPDF"
         value="D:\PDFS\CLIENTE_A"/>
  </appSettings>
  <connectionStrings>
    <add name="BD"
         connectionString="Data Source=SQLSRV01;
                           Initial Catalog=BD_CLIENTE_A;
                           User ID=usuario;
                           Password=clave;"
         providerName="System.Data.SqlClient"/>
  </connectionStrings>
</configuration>
2. Clase de configuración
using System.Configuration;
public static class Configuracion
{
    public static string Cliente
    {
        get
        {
            return ConfigurationManager.AppSettings["Cliente"];
        }
    }
    public static string NombreServicio
    {
        get
        {
            return ConfigurationManager.AppSettings["NombreServicio"];
        }
    }
    public static string DisplayName
    {
        get
        {
            return ConfigurationManager.AppSettings["DisplayName"];
        }
    }
    public static string RutaPDF
    {
        get
        {
            return ConfigurationManager.AppSettings["RutaPDF"];
        }
    }
    public static string CadenaConexion
    {
        get
        {
            return ConfigurationManager
                .ConnectionStrings["BD"]
                .ConnectionString;
        }
    }
}

3. Abrir la conexión SQL
using System.Data.SqlClient;

public void Procesar()
{
    using (SqlConnection cn =
           new SqlConnection(Configuracion.CadenaConexion))
    {
        cn.Open();

        SqlCommand cmd =
            new SqlCommand(
                "SELECT GETDATE()",
                cn);

        object valor = cmd.ExecuteScalar();
    }
}

static class Program
{
    static void Main()
    {
        ServiceBase[] ServicesToRun;

        ServicesToRun = new ServiceBase[]
        {
            new ServicioPDF()
        };

        ServiceBase.Run(ServicesToRun);
    }
}

7. Batch de instalación Cliente A
@echo off

sc create ServicioPDF_CLIENTE_A ^
binPath= "D:\Servicios\ClienteA\ServicioPDF.exe" ^
start= auto ^
DisplayName= "Servicio PDF - Cliente A"

sc description ServicioPDF_CLIENTE_A "Servicio de generacion de PDFs para Cliente A"

sc start ServicioPDF_CLIENTE_A

pause
